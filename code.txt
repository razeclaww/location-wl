  <script>
    // ==========================================
    // CONFIGURATION (EDIT BEFORE HOSTING)
    // ==========================================
    const CSR_PIN = "1234"; // Change this to your real staff PIN
    const STORAGE_KEY = "wl_saved_riders_v1";

    // Optional: set the target path/URL where customer location links point
    // Example: "https://wlfood.vercel.app/location.html" or "./location.html"
    // Leave as "" to use the current site base URL
    const TARGET_BASE_URL = "";

    // ==========================================
    // INITIALIZATION
    // ==========================================
    window.addEventListener("DOMContentLoaded", () => {
      renderSavedRiders();
      if (sessionStorage.getItem("wl_csr_unlocked") === "1") {
        document.getElementById("pinGate").classList.add("hidden");
        document.getElementById("adminDashboard").classList.remove("hidden");
      }
    });

    // ==========================================
    // PIN AUTHENTICATION
    // ==========================================
    function verifyPin() {
      const entered = document.getElementById("pinInput").value.trim();
      const errorBox = document.getElementById("pinError");

      if (entered === CSR_PIN) {
        sessionStorage.setItem("wl_csr_unlocked", "1");
        errorBox.classList.add("hidden");
        document.getElementById("pinGate").classList.add("hidden");
        document.getElementById("adminDashboard").classList.remove("hidden");
      } else {
        errorBox.classList.remove("hidden");
      }
    }

    // ==========================================
    // LINK GENERATION & RESET
    // ==========================================
    function getBaseTargetUrl() {
      if (TARGET_BASE_URL.trim()) {
        return TARGET_BASE_URL.trim();
      }
      return window.location.origin + window.location.pathname;
    }

    function makeLink() {
      const nameInput = document.getElementById("riderNameInput");
      const phoneInput = document.getElementById("riderPhoneInput");
      const errorBox = document.getElementById("formError");

      const name = nameInput.value.trim();
      const rawPhone = phoneInput.value.trim();
      const cleanPhone = rawPhone.replace(/[^0-9]/g, "");

      errorBox.classList.add("hidden");

      if (!name) {
        showFormError("Please enter the rider's name.");
        nameInput.focus();
        return;
      }

      if (cleanPhone.length < 10) {
        showFormError("Please enter a valid contact number (e.g. +9779800000000).");
        phoneInput.focus();
        return;
      }

      saveRider(name, "+" + cleanPhone);

      const baseUrl = getBaseTargetUrl();
      const separator = baseUrl.includes("?") ? "&" : "?";
      const finalUrl = `${baseUrl}${separator}r=${encodeURIComponent(name)}&p=${encodeURIComponent(cleanPhone)}`;

      document.getElementById("generatedUrl").textContent = finalUrl;
      document.getElementById("resultBox").style.display = "block";
      document.getElementById("copyBtn").textContent = "Copy Link";
    }

    function createNewLink() {
      document.getElementById("resultBox").style.display = "none";
      document.getElementById("generatedUrl").textContent = "";
      document.getElementById("formError").classList.add("hidden");
      document.getElementById("riderNameInput").value = "";
      document.getElementById("riderPhoneInput").value = "+977";
      document.getElementById("riderNameInput").focus();
    }

    function showFormError(msg) {
      const errorBox = document.getElementById("formError");
      errorBox.textContent = msg;
      errorBox.classList.remove("hidden");
    }

    function copyGeneratedLink() {
      const text = document.getElementById("generatedUrl").textContent;
      if (!text) return;

      navigator.clipboard.writeText(text).then(() => {
        const btn = document.getElementById("copyBtn");
        btn.textContent = "Copied!";
        setTimeout(() => {
          btn.textContent = "Copy Link";
        }, 2000);
      });
    }

    // ==========================================
    // SAVED RIDERS (LOCAL STORAGE)
    // ==========================================
    function getSavedRiders() {
      try {
        return JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
      } catch {
        return [];
      }
    }

    function saveRider(name, phone) {
      let list = getSavedRiders();
      list = list.filter((item) => item.phone !== phone);
      list.unshift({ name, phone });
      if (list.length > 8) list.pop();
      localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
      renderSavedRiders();
    }

    function deleteRider(e, phone) {
      e.stopPropagation();
      const list = getSavedRiders().filter((item) => item.phone !== phone);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
      renderSavedRiders();
    }

    function renderSavedRiders() {
      const list = getSavedRiders();
      const wrap = document.getElementById("savedRidersWrap");
      const container = document.getElementById("riderPills");

      if (!list.length) {
        wrap.classList.add("hidden");
        return;
      }

      wrap.classList.remove("hidden");
      container.innerHTML = "";

      list.forEach((r) => {
        const pill = document.createElement("div");
        pill.className = "pill";
        pill.innerHTML = `${r.name} <span class="del" title="Remove">&times;</span>`;
        pill.onclick = () => {
          document.getElementById("riderNameInput").value = r.name;
          document.getElementById("riderPhoneInput").value = r.phone;
          document.getElementById("formError").classList.add("hidden");
        };
        pill.querySelector(".del").onclick = (e) => deleteRider(e, r.phone);
        container.appendChild(pill);
      });
    }
  </script>
